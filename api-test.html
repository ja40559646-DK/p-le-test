export default { async fetch(request, env, ctx) { // URL'en vi vil kalde hos SDFI const url = new URL(request.url); const targetUrl = `https://api.dataforsyningen.dk${url.pathname}${url.search}`; // Tilføj CORS headers så din GitHub må læse svaret const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, HEAD, POST, OPTIONS", "Access-Control-Max-Age": "86400", }; // Hent data fra SDFI const response = await fetch(targetUrl); // Send data tilbage til din browser med de rigtige tilladelser return new Response(response.body, { status: response.status, headers: { ...response.headers, ...corsHeaders } }); }, };
